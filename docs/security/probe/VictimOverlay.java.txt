package probe.victim;
import android.app.*;
import android.content.Intent;
import android.content.pm.ServiceInfo;
import android.graphics.Color;
import android.graphics.PixelFormat;
import android.os.IBinder;
import android.util.Log;
import android.view.Gravity;
import android.view.View;
import android.view.WindowManager;
import android.widget.Button;
import android.widget.LinearLayout;

/** Mimics Akma's panel: a TYPE_APPLICATION_OVERLAY window with a normal button A and a filterTouchesWhenObscured button B. */
public class Overlay extends Service {
    private View root; private int a, b;
    @Override public IBinder onBind(Intent i) { return null; }
    @Override public int onStartCommand(Intent intent, int flags, int id) {
        NotificationChannel ch = new NotificationChannel("p", "p", NotificationManager.IMPORTANCE_LOW);
        getSystemService(NotificationManager.class).createNotificationChannel(ch);
        Notification n = new Notification.Builder(this, "p").setSmallIcon(android.R.drawable.ic_dialog_info).setContentTitle("victim probe").build();
        startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE);
        if (root != null) return START_NOT_STICKY;
        boolean secure = intent != null && "secure".equals(intent.getStringExtra("mode"));
        LinearLayout l = new LinearLayout(this); l.setOrientation(LinearLayout.VERTICAL); l.setBackgroundColor(Color.WHITE);
        Button A = new Button(this); A.setText("A plain"); A.setOnClickListener(v -> { a++; Log.i("PROBE", "victim A taps=" + a); });
        Button B = new Button(this); B.setText("B filtered"); B.setFilterTouchesWhenObscured(true); B.setOnClickListener(v -> { b++; Log.i("PROBE", "victim B taps=" + b); });
        l.addView(A, new LinearLayout.LayoutParams(600, 200)); l.addView(B, new LinearLayout.LayoutParams(600, 200)); root = l;
        int flags2 = WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL | (secure ? WindowManager.LayoutParams.FLAG_SECURE : 0);
        WindowManager.LayoutParams p = new WindowManager.LayoutParams(600, WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY, flags2, PixelFormat.TRANSLUCENT);
        p.gravity = Gravity.TOP | Gravity.CENTER_HORIZONTAL; p.y = 200;
        getSystemService(WindowManager.class).addView(root, p);
        Log.i("PROBE", "victim overlay added secure=" + secure);
        return START_NOT_STICKY;
    }
    @Override public void onDestroy() { if (root != null) getSystemService(WindowManager.class).removeView(root); root = null; super.onDestroy(); }
}
